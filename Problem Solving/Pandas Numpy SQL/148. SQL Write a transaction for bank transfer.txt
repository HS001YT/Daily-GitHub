/* TASK: Bank Transfer Transaction
   Transfer 10000 from Amit (emp_id = 1) to Riya (emp_id = 2) */

START TRANSACTION;

/* Debit amount from sender (Amit) */

UPDATE employees
SET salary = salary - 10000
WHERE emp_id = 1;

/* Credit amount to receiver (Riya) */

UPDATE employees
SET salary = salary + 10000
WHERE emp_id = 2;

/* Commit transaction */

COMMIT;
