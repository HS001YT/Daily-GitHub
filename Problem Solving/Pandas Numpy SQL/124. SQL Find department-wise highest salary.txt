/* Query to find department-wise highest salary */

SELECT emp_id, emp_name, salary, dept_id
FROM (
    SELECT emp_id,
           emp_name,
           salary,
           dept_id,
           DENSE_RANK() OVER (PARTITION BY dept_id ORDER BY salary DESC) AS rnk
    FROM employees
) t
WHERE rnk = 1;
